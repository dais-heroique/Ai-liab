Clear Governance & Legal Positioning

  • Technical Infrastructure: The Gateway enforces technical constraints and generates auditable evidence. It is not an insurance policy or legal counsel.
  • Customer Execution: The Gateway evaluates actions and outputs a deterministic decision (`ALLOW`, `BLOCK`, `HUMAN_REVIEW`). The customer’s application or orchestrator executes the tool.
  • Evidence vs Certification: The Gateway provides technical logs supporting compliance audits (e.g. EU AI Act, SOC 2), not third-party compliance rubber-stamps.
  • Fail-Closed Security: Unknown agents, malformed payloads, or policy ambiguities default strictly to `BLOCK` to prevent unauthorized execution.

TypeScript / Node.js Pre-Execution Middleware 
import axios from 'axios';

interface GatewayDecision {
  decision: 'ALLOW' | 'BLOCK' | 'HUMAN_REVIEW';
  risk_score: number;
  reason: string;
  audit_log_id: number;
  sha256_proof: string;
}

async function executeAgentToolSafely(action: {
  agentId: string;
  actionType: string;
  amount?: number;
  target?: string;
  description: string;
  parameters?: Record<string, any>;
}) {
  // 1. Evaluate with AI Liability Gateway before calling real tool
  const evalRes = await axios.post<GatewayDecision>(
    'https://gateway.ailg.internal/v1/actions/evaluate',
    {
      agent_id: action.agentId,
      action_type: action.actionType,
      amount: action.amount,
      target: action.target,
      description: action.description,
      parameters: action.parameters,
      idempotency_key: `idemp_${Date.now()}`
    },
    {
      headers: { Authorization: `Bearer ${process.env.AILG_API_KEY}` }
    }
  );

  const evaluation = evalRes.data;

  // 2. Deterministic outcome check
  if (evaluation.decision === 'BLOCK') {
    throw new Error(`Execution Blocked by Gateway: ${evaluation.reason}`);
  }

  if (evaluation.decision === 'HUMAN_REVIEW') {
    console.log(`Action held in escrow for human review: Log #${evaluation.audit_log_id}`);
    return { status: 'PENDING_HUMAN_APPROVAL', auditId: evaluation.audit_log_id };
  }

  // 3. Dispatch actual tool if ALLOW
  console.log(`Gateway ALLOW: Risk Score ${evaluation.risk_score}. Executing tool...`);
  const toolResult = await dispatchStripeRefund(action.parameters);

  // 4. Record execution proof
  await axios.post(
    'https://gateway.ailg.internal/v1/actions/execute',
    { audit_log_id: evaluation.audit_log_id, status: 'SUCCESS' },
    { headers: { Authorization: `Bearer ${process.env.AILG_API_KEY}` } }
  );

  return toolResult;
}

import os
import requests

AILG_ENDPOINT = "https://gateway.ailg.internal/v1/actions/evaluate"
AILG_API_KEY = os.environ.get("AILG_API_KEY", "gw_live_8f9021a8cd34e10b")

def guard_agent_action(agent_id: str, action_type: str, amount: float = None, description: str = "") -> dict:
    """Pre-action evaluation gateway call."""
    headers = {
        "Authorization": f"Bearer {AILG_API_KEY}",
        "Content-Type": "application/json"
    }
    payload = {
        "agent_id": agent_id,
        "action_type": action_type,
        "amount": amount,
        "description": description
    }
    
    response = requests.post(AILG_ENDPOINT, json=payload, headers=headers, timeout=5)
    response.raise_for_status()
    result = response.json()
    
    # Check decision
    if result["decision"] == "BLOCK":
        raise PermissionError(f"Gateway Blocked Action: {result['reason']}")
    elif result["decision"] == "HUMAN_REVIEW":
        print(f"Action held in escrow (Escalation ID: {result.get('escalation_id')})")
        return {"status": "ESCALATED", "details": result}
    
    print(f"Action APPROVED with Risk Score {result['risk_score']}")
    return result
Express.js Webhook Receiver 
import express from 'express';
const app = express();
app.use(express.json());

// Receive Gateway real-time lifecycle webhooks
app.post('/webhooks/ailg-gateway', (req, res) => {
  const { event_type, audit_log, data, timestamp } = req.body;

  switch (event_type) {
    case 'action.review_required':
      console.log(`[ALERT] Action escalated to human review: Log #${audit_log.id}`);
      notifySlackSecurityChannel(data.escalation_id, audit_log.agent_id, audit_log.description);
      break;

    case 'action.approved':
      console.log(`[RESOLVED] Action approved by operator ${data.operator}. Dispatching tool.`);
      resumePendingAgentAction(audit_log.id);
      break;

    case 'action.blocked':
      console.warn(`[SECURITY] Action blocked: Policy '${data.policy_triggered}'`);
      break;
  }

  res.status(200).json({ received: true });
});

app.listen(8080);
